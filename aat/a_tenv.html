NOVALUE:
  say 'Novalue raised in line' Sigl
  say SOURCELINE(Sigl)
  say 'Variable' CONDITION('D')
  signal LOOKAROUND
 
SYNTAX:
  say 'Syntax raised in line' Sigl
  say SOURCELINE(Sigl)
  say 'rc='rc '('ERRORTEXT(rc)')'
 
HALT:
LOOKAROUND:
  do
    say 'You can look around now.'
    trace ?R
    nop
  end
exit 12
